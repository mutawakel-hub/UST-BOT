    26→    26→    26→## 1. نظرة عامة على المشروع
    27→    27→    27→
    28→    28→    28→### 1.1 الهدف
    29→    29→    29→بناء نظام بوت تلغرام مركزي لجامعة العلوم والتكنولوجيا، يخدم آلاف الطلاب للوصول إلى المحتوى الأكاديمي (مقررات، نماذج، ملخصات)، مع نظام مساهمات طلابي ولوحة إدارة هرمية من 4 مستويات.
    30→    30→    30→
    31→    31→    31→### 1.2 المكوّنات الرئيسية
    32→    32→    32→
    33→    33→    33→| المكوّن | الوصف |
    34→    34→    34→|---|---|
    35→    35→    35→| 🎓 **بوت الطالب** | 12 شاشة أساسية لتصفح المحتوى والمساهمة والاشتراكات |
    36→    36→    36→| 🛡 **بوت الإدارة** | 15 شاشة إدارية مع 4 أدوار هرمية |
    37→    37→    37→| 🗄 **قاعدة البيانات** | Supabase (PostgreSQL) — 10 جداول أساسية |
    38→    38→    38→| 📦 **التخزين** | 7 قنوات تلغرام خاصة (قناة لكل كلية) |
    39→    39→    39→| ⚡ **الـ Runtime** | Cloudflare Workers (مجاني 100%) |
    40→    40→    40→
    41→    41→    41→### 1.3 المبادئ التصميمية
    42→    42→    42→
    43→    43→    43→- **صفر تكلفة:** جميع الخدمات ضمن الباقات المجانية
    44→    44→    44→- **أداء عالٍ:** استجابة < 500ms لكل طلب
    45→    45→    45→- **أمان مزدوج:** فحص الصلاحية في الواجهة والباك إند
    46→    46→    46→- **تخصيص كامل:** جميع النصوص والأزرار قابلة للتعديل من الإدارة
    47→    47→    47→- **بساطة الطالب:** تدفق مباشر من 6 خطوات للوصول لأي ملف

    51→    51→    51→## 2. التقنيات المؤكدة
    52→    52→    52→
    53→    53→    53→| الطبقة | التقنية | السبب |

    60→    60→    60→| طوابير الإرسال | **Cloudflare Queues** | للتعميمات الجماعية |

   414→   414→   414→-- 8. المسؤولون (4 أدوار هرمية)

   417→   417→   417→  id SERIAL PRIMARY KEY,
   418→   418→   418→  telegram_id BIGINT UNIQUE NOT NULL,
   419→   419→   419→  name TEXT NOT NULL,
   420→   420→   420→  role TEXT NOT NULL CHECK (role IN ('central', 'college', 'specialty', 'level')),
   421→   421→   421→  college_id INT REFERENCES colleges(id),         -- لـ college/specialty/level
   422→   422→   422→  specialty_id INT REFERENCES specialties(id),    -- لـ specialty/level
   423→   423→   423→  level INT,                                       -- لـ level فقط

   425→   425→   425→  created_at TIMESTAMP DEFAULT NOW(),

   427→   427→   427→  CONSTRAINT chk_admin_scope CHECK (
   428→   428→   428→    (role = 'central' AND college_id IS NULL AND specialty_id IS NULL AND level IS NULL) OR
   429→   429→   429→    (role = 'college' AND college_id IS NOT NULL AND specialty_id IS NULL AND level IS NULL) OR
   430→   430→   430→    (role = 'specialty' AND college_id IS NOT NULL AND specialty_id IS NOT NULL AND level IS NULL) OR
   431→   431→   431→    (role = 'level' AND college_id IS NOT NULL AND specialty_id IS NOT NULL AND level IS NOT NULL)
   432→   432→   432→  )
   433→   433→   433→);
   434→   434→   434→CREATE INDEX idx_admins_telegram ON admins(telegram_id);

   461→   461→   461→  updated_at TIMESTAMP DEFAULT NOW(),

   487→   487→   487→  created_at TIMESTAMP DEFAULT NOW()
   488→   488→   488→);
   489→   489→   489→CREATE INDEX idx_downloads_user ON downloads(user_id, created_at DESC);
   490→   490→   490→CREATE INDEX idx_downloads_file ON downloads(file_id);
   491→   491→   491→
   492→   492→   492→-- ============================================
   493→   493→   493→-- 13. التعميمات (للسجل + منع التكرار)

   496→   496→   496→  id BIGSERIAL PRIMARY KEY,
   497→   497→   497→  admin_id INT NOT NULL REFERENCES admins(id),
   498→   498→   498→  scope_type TEXT NOT NULL CHECK (scope_type IN ('all', 'college', 'specialty', 'level')),
   499→   499→   499→  scope_college_id INT,
   500→   500→   500→  scope_specialty_id INT,
   501→   501→   501→  scope_level INT,
   502→   502→   502→  content_type TEXT NOT NULL CHECK (content_type IN ('text', 'photo', 'document')),
   503→   503→   503→  text_content TEXT,
   504→   504→   504→  media_file_id TEXT,
   505→   505→   505→  sent_count INT DEFAULT 0,
   506→   506→   506→  failed_count INT DEFAULT 0,
   507→   507→   507→  status TEXT DEFAULT 'queued' CHECK (status IN ('queued', 'sending', 'completed', 'failed')),
   508→   508→   508→  created_at TIMESTAMP DEFAULT NOW()
   509→   509→   509→);
   510→   510→   510→
   511→   511→   511→-- ============================================

   514→   514→   514→CREATE TABLE leaderboard (
   515→   515→   515→  id BIGSERIAL PRIMARY KEY,
   516→   516→   516→  user_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
   517→   517→   517→  scope_type TEXT NOT NULL CHECK (scope_type IN ('global', 'college', 'specialty')),
   518→   518→   518→  scope_college_id INT,
   519→   519→   519→  scope_specialty_id INT,
   520→   520→   520→  rank INT NOT NULL,
   521→   521→   521→  points INT NOT NULL,
   522→   522→   522→  period_start DATE,
   523→   523→   523→  period_end DATE,
   524→   524→   524→  updated_at TIMESTAMP DEFAULT NOW(),
   525→   525→   525→  UNIQUE(user_id, scope_type, scope_college_id, scope_specialty_id, period_start)
   526→   526→   526→);
   527→   527→   527→CREATE INDEX idx_leaderboard_scope ON leaderboard(scope_type, scope_college_id, scope_specialty_id, rank);
